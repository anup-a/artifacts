# artifacts server Implementation Plan (plan 1 of 3)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A single Go binary that stores content-addressed artifact versions, serves them at `/a/<id>/…` with the comments runtime injected, and exposes the publisher API and viewer comment API from the spec.

**Architecture:** `net/http` with Go 1.22 pattern routing; one `store.Store` interface backed by SQLite (dev/self-host) or Postgres (managed) through `database/sql`; one `blob.Store` interface backed by local disk or any S3-compatible bucket; handlers return typed `apierr.Error` values mapped to the spec's error table. Everything embedded (migrations, runtime JS placeholder), so the binary runs from a bare `docker run`.

**Tech Stack:** Go 1.22+, `modernc.org/sqlite` (CGO-free), `github.com/jackc/pgx/v5/stdlib`, `github.com/minio/minio-go/v7`, stdlib `testing` + `net/http/httptest`.

Spec: `docs/superpowers/specs/2026-09-15-artifacts-service-design.md`. Plans 2 (runtime) and 3 (CLI + skill + deploy) follow.

## Global Constraints

- Module path `github.com/anup-a/byagent` (rename with the product name later; one `sed`).
- Go ≥ 1.22 (pattern routing, `for range int`).
- No CGO. `modernc.org/sqlite` only.
- Artifact id: 12 chars base62 from 9 random bytes. Key: `ak_` + 32 random bytes base62; only SHA-256 hex stored.
- Error envelope: `{"error":{"code":"…","message":"…","request_id":"…"}}`; every response has `X-Request-Id`.
- Codes and statuses: `INVALID_ARGUMENT` 400, `UNAUTHORIZED` 401, `FORBIDDEN` 403, `NOT_FOUND` 404, `GONE` 410, `PAYLOAD_TOO_LARGE` 413, `RATE_LIMITED` 429, `QUOTA_EXCEEDED` 429, `INTERNAL` 500.
- Default limits (env-overridable): `LIMIT_VERSION_BYTES=26214400`, `LIMIT_FILES=500`, `LIMIT_VERSIONS_KEPT=20`, `LIMIT_PUBLISH_PER_MIN=10`, `LIMIT_ARTIFACTS_PER_KEY=200`, `LIMIT_COMMENT_BYTES=2048`, `LIMIT_COMMENTS_PER_MIN=20`, `LIMIT_ANCHOR_BYTES=4096`.
- Blob key layout: `blobs/<sha256[0:2]>/<sha256>`.
- Every publish requires `index.html` at the manifest root.
- Commit after every task with a conventional-commit message. Never modify files through Bash.

```go
// Package config reads server configuration from environment variables.
package config

import (
	"errors"
	"fmt"
	"strconv"
)

type Limits struct {
	VersionBytes    int64
	Files           int
	VersionsKept    int
	PublishPerMin   int
	ArtifactsPerKey int
	CommentBytes    int
	CommentsPerMin  int
	AnchorBytes     int
}

type Config struct {
	Addr        string
	DB          string // sqlite | postgres
	DatabaseURL string
	Storage     string // local | s3
	DataDir     string
	S3Endpoint  string
	S3Bucket    string
	S3AccessKey string
	S3SecretKey string
	S3UseSSL    bool
	AdminToken  string
	KeySecret   string
	ShareOrigin string
	APIOrigin   string
	Limits      Limits
}

func FromEnv(getenv func(string) string) (Config, error) {
	get := func(k, def string) string {
		if v := getenv(k); v != "" {
			return v
		}
		return def
	}
	c := Config{
		Addr:        get("ADDR", ":8080"),
		DB:          get("DB", "sqlite"),
		DatabaseURL: getenv("DATABASE_URL"),
		Storage:     get("STORAGE", "local"),
		DataDir:     get("DATA_DIR", "./data"),
		S3Endpoint:  getenv("S3_ENDPOINT"),
		S3Bucket:    getenv("S3_BUCKET"),
		S3AccessKey: getenv("S3_ACCESS_KEY"),
		S3SecretKey: getenv("S3_SECRET_KEY"),
		S3UseSSL:    get("S3_USE_SSL", "true") == "true",
		AdminToken:  getenv("ADMIN_TOKEN"),
		KeySecret:   getenv("KEY_SIGNING_SECRET"),
		ShareOrigin: get("PUBLIC_SHARE_ORIGIN", "http://localhost:8080"),
		APIOrigin:   get("PUBLIC_API_ORIGIN", "http://localhost:8080"),
	}
	var err error
	if c.Limits, err = limitsFromEnv(getenv); err != nil {
		return c, err
	}
	switch c.DB {
	case "sqlite":
	case "postgres":
		if c.DatabaseURL == "" {
			return c, errors.New("DB=postgres requires DATABASE_URL")
		}
	default:
		return c, fmt.Errorf("DB must be sqlite or postgres, got %q", c.DB)
	}
	switch c.Storage {
	case "local":
	case "s3":
		if c.S3Endpoint == "" || c.S3Bucket == "" || c.S3AccessKey == "" || c.S3SecretKey == "" {
			return c, errors.New("STORAGE=s3 requires S3_ENDPOINT, S3_BUCKET, S3_ACCESS_KEY, S3_SECRET_KEY")
		}
	default:
		return c, fmt.Errorf("STORAGE must be local or s3, got %q", c.Storage)
	}
	if c.AdminToken == "" && c.KeySecret == "" {
		return c, errors.New("set ADMIN_TOKEN (self-host) or KEY_SIGNING_SECRET (managed)")
	}
	return c, nil
}

func limitsFromEnv(getenv func(string) string) (Limits, error) {
	l := Limits{VersionBytes: 26214400, Files: 500, VersionsKept: 20, PublishPerMin: 10,
		ArtifactsPerKey: 200, CommentBytes: 2048, CommentsPerMin: 20, AnchorBytes: 4096}
	set := func(k string, dst *int) error {
		v := getenv(k)
		if v == "" {
			return nil
		}
		n, err := strconv.Atoi(v)
		if err != nil || n <= 0 {
			return fmt.Errorf("%s must be a positive integer, got %q", k, v)
		}
		*dst = n
		return nil
	}
	if v := getenv("LIMIT_VERSION_BYTES"); v != "" {
		n, err := strconv.ParseInt(v, 10, 64)
		if err != nil || n <= 0 {
			return l, fmt.Errorf("LIMIT_VERSION_BYTES must be a positive integer, got %q", v)
		}
		l.VersionBytes = n
	}
	for k, dst := range map[string]*int{
		"LIMIT_FILES": &l.Files, "LIMIT_VERSIONS_KEPT": &l.VersionsKept,
		"LIMIT_PUBLISH_PER_MIN": &l.PublishPerMin, "LIMIT_ARTIFACTS_PER_KEY": &l.ArtifactsPerKey,
		"LIMIT_COMMENT_BYTES": &l.CommentBytes, "LIMIT_COMMENTS_PER_MIN": &l.CommentsPerMin,
		"LIMIT_ANCHOR_BYTES": &l.AnchorBytes,
	} {
		if err := set(k, dst); err != nil {
			return l, err
		}
	}
	return l, nil
}
```

```go
// Package apierr defines the API error envelope and typed error constructors.
package apierr

import (
	"context"
	"encoding/json"
	"errors"
	"fmt"
	"log/slog"
	"net/http"
	"strconv"
)

type Error struct {
	Code       string
	Status     int
	Message    string
	RetryAfter int
	cause      error
}

func (e *Error) Error() string { return e.Code + ": " + e.Message }
func (e *Error) Unwrap() error { return e.cause }

func Invalid(msg string) *Error      { return &Error{"INVALID_ARGUMENT", 400, msg, 0, nil} }
func Unauthorized() *Error           { return &Error{"UNAUTHORIZED", 401, "missing or invalid API key", 0, nil} }
func Forbidden() *Error              { return &Error{"FORBIDDEN", 403, "not allowed", 0, nil} }
func NotFound(what string) *Error    { return &Error{"NOT_FOUND", 404, what + " not found", 0, nil} }
func Gone() *Error                   { return &Error{"GONE", 410, "artifact deleted", 0, nil} }
func TooLarge(msg string) *Error     { return &Error{"PAYLOAD_TOO_LARGE", 413, msg, 0, nil} }
func RateLimited(sec int) *Error     { return &Error{"RATE_LIMITED", 429, "rate limited", sec, nil} }
func Quota(msg string) *Error        { return &Error{"QUOTA_EXCEEDED", 429, msg, 0, nil} }
func Internal(err error) *Error      { return &Error{"INTERNAL", 500, "internal error", 0, err} }

type ctxKey struct{}

func WithRequestID(ctx context.Context, id string) context.Context {
	return context.WithValue(ctx, ctxKey{}, id)
}

func RequestID(ctx context.Context) string {
	id, _ := ctx.Value(ctxKey{}).(string)
	return id
}

type envelope struct {
	Error struct {
		Code      string `json:"code"`
		Message   string `json:"message"`
		RequestID string `json:"request_id"`
	} `json:"error"`
}

// Write maps err to the envelope. Non-*Error values become INTERNAL and are logged.
func Write(w http.ResponseWriter, r *http.Request, err error) {
	var e *Error
	if !errors.As(err, &e) {
		e = Internal(err)
	}
	if e.Status == 500 {
		slog.Error("internal error", "request_id", RequestID(r.Context()), "err", fmt.Sprint(err))
	}
	if e.RetryAfter > 0 {
		w.Header().Set("Retry-After", strconv.Itoa(e.RetryAfter))
	}
	var env envelope
	env.Error.Code, env.Error.Message, env.Error.RequestID = e.Code, e.Message, RequestID(r.Context())
	WriteJSON(w, e.Status, env)
}

func WriteJSON(w http.ResponseWriter, status int, v any) {
	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(status)
	_ = json.NewEncoder(w).Encode(v)
}
```

```go
// Package ids generates artifact ids and API keys.
package ids

import (
	"crypto/rand"
	"crypto/sha256"
	"encoding/hex"
	"math/big"
)

const alphabet = "0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz"

func base62(b []byte, width int) string {
	n := new(big.Int).SetBytes(b)
	base := big.NewInt(62)
	out := make([]byte, width)
	for i := width - 1; i >= 0; i-- {
		var m big.Int
		n.DivMod(n, base, &m)
		out[i] = alphabet[m.Int64()]
	}
	return string(out)
}

func random(n int) []byte {
	b := make([]byte, n)
	if _, err := rand.Read(b); err != nil {
		panic(err) // crypto/rand failure is unrecoverable
	}
	return b
}

// NewArtifactID returns 12 base62 chars from 9 random bytes.
func NewArtifactID() string { return base62(random(9), 12) }

// NewKey returns the plaintext key (shown once) and its sha256 hex (stored).
func NewKey() (string, string) {
	plain := "ak_" + base62(random(32), 43)
	return plain, HashKey(plain)
}

func HashKey(plain string) string {
	h := sha256.Sum256([]byte(plain))
	return hex.EncodeToString(h[:])
}
```

```sql
  artifact_id TEXT NOT NULL REFERENCES artifacts(id),

  created_at TIMESTAMP NOT NULL,
  bytes BIGINT NOT NULL,

  PRIMARY KEY (artifact_id, version)
);

CREATE TABLE IF NOT EXISTS files (
  artifact_id TEXT NOT NULL,
```

```go
//go:embed migrations/*.sql
var migrationFS embed.FS

func runMigrations(ctx context.Context, db *sql.DB) error {
	if _, err := db.ExecContext(ctx, `CREATE TABLE IF NOT EXISTS schema_migrations (version INTEGER PRIMARY KEY)`); err != nil {
		return err
	}
	entries, err := migrationFS.ReadDir("migrations")
	if err != nil {
		return err
	}
	names := make([]string, 0, len(entries))
	for _, e := range entries {
		names = append(names, e.Name())
	}
	sort.Strings(names)
	for _, name := range names {
		ver, err := strconv.Atoi(strings.SplitN(name, "_", 2)[0])
		if err != nil {
			return fmt.Errorf("migration name %q: %w", name, err)
		}
		var n int
		if err := db.QueryRowContext(ctx, `SELECT COUNT(*) FROM schema_migrations WHERE version = ?`, ver).Scan(&n); err != nil {
			return err
		}
		if n > 0 {
			continue
		}
		body, err := migrationFS.ReadFile("migrations/" + name)
		if err != nil {
			return err
		}
		tx, err := db.BeginTx(ctx, nil)
		if err != nil {
			return err
		}
		if _, err := tx.ExecContext(ctx, string(body)); err != nil {
			tx.Rollback()
			return fmt.Errorf("migration %s: %w", name, err)
		}
		if _, err := tx.ExecContext(ctx, `INSERT INTO schema_migrations(version) VALUES (?)`, ver); err != nil {
			tx.Rollback()
			return err
		}
		if err := tx.Commit(); err != nil {
			return err
		}
	}
	return nil
}
```

```go
type sqlStore struct {
	db      *sql.DB
	dialect string
}

// q rewrites ? placeholders to $1.. for postgres.
func (s *sqlStore) q(query string) string {
	if s.dialect != "postgres" {
		return query
	}
	var b strings.Builder
	n := 0
	for _, r := range query {
		if r == '?' {
			n++
			b.WriteString("$" + strconv.Itoa(n))
		} else {
			b.WriteRune(r)
		}
	}
	return b.String()
}

func (s *sqlStore) PruneVersions(ctx context.Context, artifactID string, keep int) ([]int, error) {
	vs, err := s.Versions(ctx, artifactID)
	if err != nil {
		return nil, err
	}
	if len(vs) <= keep {
		return nil, nil
	}
	protected := map[int]bool{}
	rows, err := s.db.QueryContext(ctx, s.q(`SELECT DISTINCT version FROM threads WHERE artifact_id=? AND status='open'`), artifactID)
	if err != nil {
		return nil, err
	}
	for rows.Next() {
		var v int
		if err := rows.Scan(&v); err != nil {
			rows.Close()
			return nil, err
		}
		protected[v] = true
	}
	rows.Close()
	excess := len(vs) - keep
	var pruned []int
	tx, err := s.db.BeginTx(ctx, nil)
	if err != nil {
		return nil, err
	}
	defer tx.Rollback()
	for _, v := range vs {
		if excess == 0 {
			break
		}
		if protected[v.Version] {
			continue
		}
		if _, err := tx.ExecContext(ctx, s.q(`UPDATE blobs SET refcount=refcount-1 WHERE sha256 IN (SELECT sha256 FROM files WHERE artifact_id=? AND version=?)`), artifactID, v.Version); err != nil {
			return nil, err
		}
		if _, err := tx.ExecContext(ctx, s.q(`DELETE FROM files WHERE artifact_id=? AND version=?`), artifactID, v.Version); err != nil {
			return nil, err
		}
		if _, err := tx.ExecContext(ctx, s.q(`DELETE FROM versions WHERE artifact_id=? AND version=?`), artifactID, v.Version); err != nil {
			return nil, err
		}
		pruned = append(pruned, v.Version)
		excess--
	}
	return pruned, tx.Commit()
}

func (s *sqlStore) UpsertBlob(ctx context.Context, sha string, size int64) error {
	var q string
	if s.dialect == "postgres" {
		q = `INSERT INTO blobs(sha256,size,refcount,created_at) VALUES (?,?,0,?) ON CONFLICT (sha256) DO NOTHING`
	} else {
		q = `INSERT OR IGNORE INTO blobs(sha256,size,refcount,created_at) VALUES (?,?,0,?)`
	}
	_, err := s.db.ExecContext(ctx, s.q(q), sha, size, time.Now().UTC())
	return err
}

func (s *sqlStore) HasBlobs(ctx context.Context, shas []string) (map[string]bool, error) {
	out := map[string]bool{}
	if len(shas) == 0 {
		return out, nil
	}
	ph := strings.TrimSuffix(strings.Repeat("?,", len(shas)), ",")
	args := make([]any, len(shas))
	for i, v := range shas {
		args[i] = v
	}
	rows, err := s.db.QueryContext(ctx, s.q(`SELECT sha256 FROM blobs WHERE sha256 IN (`+ph+`)`), args...)
	if err != nil {
		return nil, err
	}
	defer rows.Close()
	for rows.Next() {
		var sha string
		if err := rows.Scan(&sha); err != nil {
			return nil, err
		}
		out[sha] = true
	}
	return out, rows.Err()
}

func (s *sqlStore) ZeroRefBlobs(ctx context.Context, olderThan time.Time) ([]string, error) {

	if err != nil {
		return nil, err
	}
	defer rows.Close()
	var out []string
	for rows.Next() {
		var sha string
		if err := rows.Scan(&sha); err != nil {
			return nil, err
```

```go
package blob

import (
	"context"
	"crypto/sha256"
	"encoding/hex"
	"errors"
	"io"
	"os"
	"path/filepath"
)

type local struct{ root string }

func newLocal(root string) (*local, error) {
	if err := os.MkdirAll(filepath.Join(root, "blobs"), 0o755); err != nil {
		return nil, err
	}
	return &local{root: root}, nil
}

func (l *local) path(sha string) string { return filepath.Join(l.root, filepath.FromSlash(Key(sha))) }

func (l *local) Put(_ context.Context, sha string, r io.Reader, size int64) error {
	dst := l.path(sha)
	if _, err := os.Stat(dst); err == nil {
		return nil
	}
	if err := os.MkdirAll(filepath.Dir(dst), 0o755); err != nil {
		return err
	}
	tmp, err := os.CreateTemp(filepath.Dir(dst), ".tmp-*")
	if err != nil {
		return err
	}
	defer os.Remove(tmp.Name())
	h := sha256.New()
	if _, err := io.Copy(io.MultiWriter(tmp, h), io.LimitReader(r, size)); err != nil {
		tmp.Close()
		return err
	}
	if err := tmp.Close(); err != nil {
		return err
	}
	if hex.EncodeToString(h.Sum(nil)) != sha {
		return ErrChecksum
	}
	return os.Rename(tmp.Name(), dst)
}

func (l *local) Get(_ context.Context, sha string) (io.ReadCloser, int64, error) {
	f, err := os.Open(l.path(sha))
	if errors.Is(err, os.ErrNotExist) {
		return nil, 0, ErrNotFound
	}
	if err != nil {
		return nil, 0, err
	}
	st, err := f.Stat()
	if err != nil {
		f.Close()
		return nil, 0, err
	}
	return f, st.Size(), nil
}

func (l *local) Exists(_ context.Context, sha string) (bool, error) {
	_, err := os.Stat(l.path(sha))
	if errors.Is(err, os.ErrNotExist) {
		return false, nil
	}
	return err == nil, err
}

func (l *local) Delete(_ context.Context, sha string) error {
	err := os.Remove(l.path(sha))
	if errors.Is(err, os.ErrNotExist) {
		return nil
	}
	return err
}

func (l *local) Ping(_ context.Context) error {
	_, err := os.Stat(filepath.Join(l.root, "blobs"))
	return err
}
```

```go
	bucket string

func newS3(endpoint, bucket, access, secret string, ssl bool) (*s3store, error) {
	c, err := minio.New(endpoint, &minio.Options{Creds: credentials.NewStaticV4(access, secret, ""), Secure: ssl})
	if err != nil {
		return nil, err
	}
	return &s3store{c: c, bucket: bucket}, nil
}

func (s *s3store) Put(ctx context.Context, sha string, r io.Reader, size int64) error {
	if ok, err := s.Exists(ctx, sha); err != nil || ok {
		return err
	}
	// Buffering keeps checksum verification simple; versions are capped at LIMIT_VERSION_BYTES.
	data, err := io.ReadAll(io.LimitReader(r, size))
	if err != nil {
		return err
	}
	h := sha256.Sum256(data)
	if hex.EncodeToString(h[:]) != sha {
		return ErrChecksum
	}
	_, err = s.c.PutObject(ctx, s.bucket, Key(sha), bytesReader(data), int64(len(data)), minio.PutObjectOptions{ContentType: "application/octet-stream"})
	return err
}

func (s *s3store) Get(ctx context.Context, sha string) (io.ReadCloser, int64, error) {
	obj, err := s.c.GetObject(ctx, s.bucket, Key(sha), minio.GetObjectOptions{})
	if err != nil {
		return nil, 0, err
	}
	st, err := obj.Stat()
	if err != nil {
		obj.Close()
		var resp minio.ErrorResponse
		if errors.As(err, &resp) && resp.Code == "NoSuchKey" {
			return nil, 0, ErrNotFound
		}
		return nil, 0, err
	}
	return obj, st.Size, nil
}

func (s *s3store) Exists(ctx context.Context, sha string) (bool, error) {
	_, err := s.c.StatObject(ctx, s.bucket, Key(sha), minio.StatObjectOptions{})
	if err == nil {
		return true, nil
	}
	var resp minio.ErrorResponse
	if errors.As(err, &resp) && resp.Code == "NoSuchKey" {
		return false, nil
	}
	return false, err
}

func (s *s3store) Delete(ctx context.Context, sha string) error {
	return s.c.RemoveObject(ctx, s.bucket, Key(sha), minio.RemoveObjectOptions{})
}

func (s *s3store) Ping(ctx context.Context) error {
	_, err := s.c.BucketExists(ctx, s.bucket)
	return err
```

```go
// Package ratelimit is a fixed-window per-key limiter for a single process.
package ratelimit

import (
	"sync"
	"time"
)

type entry struct {
	windowStart time.Time
	count       int
}

type Limiter struct {
	mu    sync.Mutex
	per   int
	now   func() time.Time
	items map[string]*entry
	sweep time.Time
}

func New(perMinute int) *Limiter {
	return &Limiter{per: perMinute, now: time.Now, items: map[string]*entry{}}
}

func (l *Limiter) SetNow(f func() time.Time) { l.now = f }

func (l *Limiter) Allow(key string) (bool, int) {
	l.mu.Lock()
	defer l.mu.Unlock()
	now := l.now()
	if now.Sub(l.sweep) > time.Minute {
		for k, e := range l.items {
			if now.Sub(e.windowStart) > time.Minute {
				delete(l.items, k)
			}
		}
		l.sweep = now
	}
	e := l.items[key]
	if e == nil || now.Sub(e.windowStart) >= time.Minute {
		l.items[key] = &entry{windowStart: now, count: 1}
		return true, 0
	}
	if e.count < l.per {
		e.count++
		return true, 0
	}
	retry := int(time.Minute - now.Sub(e.windowStart)) / int(time.Second)
	if retry < 1 {
		retry = 1
	}
	return false, retry
}
```

```go
func requestID(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		b := make([]byte, 8)
		rand.Read(b)
		id := hex.EncodeToString(b)
		w.Header().Set("X-Request-Id", id)
		next.ServeHTTP(w, r.WithContext(apierr.WithRequestID(r.Context(), id)))
	})
}

func (s *server) withAuth(next http.HandlerFunc) http.HandlerFunc {
	return func(w http.ResponseWriter, r *http.Request) {
		tok := strings.TrimPrefix(r.Header.Get("Authorization"), "Bearer ")
		if tok == "" || tok == r.Header.Get("Authorization") {
			apierr.Write(w, r, apierr.Unauthorized())
			return
		}
		var k store.Key
		if s.adminKey != nil {
			if tok != s.Cfg.AdminToken {
				apierr.Write(w, r, apierr.Unauthorized())
				return
			}
			k = *s.adminKey
		} else {
			got, err := s.Store.KeyByHash(r.Context(), ids.HashKey(tok))
			if err != nil || got.RevokedAt != nil {
				apierr.Write(w, r, apierr.Unauthorized())
				return
			}
			k = got
		}
		next(w, r.WithContext(withKey(r.Context(), k)))
	}
}

// clientIP honours Cloudflare / proxy headers, falling back to RemoteAddr.
func clientIP(r *http.Request) string {
	if v := r.Header.Get("CF-Connecting-IP"); v != "" {
		return v
	}
	if v := r.Header.Get("X-Forwarded-For"); v != "" {
		return strings.TrimSpace(strings.Split(v, ",")[0])
	}
	host := r.RemoteAddr
	if i := strings.LastIndex(host, ":"); i > 0 {
		host = host[:i]
	}
	return host
}
```

```go
package api

import (
	"encoding/json"
	"fmt"
	"mime"
	"path"
	"regexp"
	"strings"

	"github.com/anup-a/byagent/internal/apierr"
	"github.com/anup-a/byagent/internal/config"
)

type manifestFile struct {
	Path   string `json:"path"`
	SHA256 string `json:"sha256"`
	Size   int64  `json:"size"`
}

type manifest struct {
	Title string         `json:"title"`
	Files []manifestFile `json:"files"`
}

var shaRe = regexp.MustCompile(`^[0-9a-f]{64}$`)

func parseManifest(raw []byte, lim config.Limits) (manifest, *apierr.Error) {
	var m manifest
	if err := json.Unmarshal(raw, &m); err != nil {
		return m, apierr.Invalid("manifest is not valid JSON")
	}
	if len(m.Title) > 200 {
		return m, apierr.Invalid("title exceeds 200 characters")
	}
	if len(m.Files) == 0 {
		return m, apierr.Invalid("manifest has no files")
	}
	if len(m.Files) > lim.Files {
		return m, apierr.TooLarge(fmt.Sprintf("too many files: %d > %d", len(m.Files), lim.Files))
	}
	seen := map[string]bool{}
	var total int64
	hasIndex := false
	for i, f := range m.Files {
		if err := validPath(f.Path); err != "" {
			return m, apierr.Invalid(fmt.Sprintf("files[%d].path: %s", i, err))
		}
		if seen[f.Path] {
			return m, apierr.Invalid("duplicate path " + f.Path)
		}
		seen[f.Path] = true
		if !shaRe.MatchString(f.SHA256) {
			return m, apierr.Invalid(fmt.Sprintf("files[%d].sha256 must be 64 lowercase hex", i))
		}
		if f.Size < 0 {
			return m, apierr.Invalid(fmt.Sprintf("files[%d].size negative", i))
		}
		total += f.Size
		if f.Path == "index.html" {
			hasIndex = true
		}
	}
	if total > lim.VersionBytes {
		return m, apierr.TooLarge(fmt.Sprintf("version is %d bytes, limit %d", total, lim.VersionBytes))
	}
	if !hasIndex {
		return m, apierr.Invalid("index.html is required at the root")
	}
	return m, nil
}

func validPath(p string) string {
	if p == "" || len(p) > 512 {
		return "empty or too long"
	}
	if strings.HasPrefix(p, "/") || strings.Contains(p, "\\") {
		return "must be relative with forward slashes"
	}
	if path.Clean(p) != p || strings.HasPrefix(p, "../") || p == ".." || strings.Contains(p, "/../") {
		return "must be a clean path without .."
	}
	return ""
}

func contentTypeFor(p string) string {
	switch strings.ToLower(path.Ext(p)) {
	case ".html", ".htm":
		return "text/html; charset=utf-8"
	case ".css":
		return "text/css; charset=utf-8"
	case ".js", ".mjs":
		return "text/javascript; charset=utf-8"
	case ".json":
		return "application/json"
	case ".svg":
		return "image/svg+xml"
	case ".md", ".txt":
		return "text/plain; charset=utf-8"
	}
	if ct := mime.TypeByExtension(path.Ext(p)); ct != "" {
		return ct
	}
	return "application/octet-stream"
}
```

```go
func (s *server) ownedArtifact(r *http.Request, id string) (store.Artifact, *apierr.Error) {

func (s *server) preflight(w http.ResponseWriter, r *http.Request) {
	raw, err := io.ReadAll(http.MaxBytesReader(w, r.Body, 1<<20))
	if err != nil {
		apierr.Write(w, r, apierr.TooLarge("manifest too large"))
		return
	}
	m, e := parseManifest(raw, s.Cfg.Limits)
	if e != nil {
		apierr.Write(w, r, e)
		return
	}
	missing, e := s.missingBlobs(r.Context(), m)
	if e != nil {
		apierr.Write(w, r, e)
		return
	}
	apierr.WriteJSON(w, 200, map[string]any{"missing": missing})
}

func (s *server) missingBlobs(ctx context.Context, m manifest) ([]string, *apierr.Error) {
	shas := make([]string, 0, len(m.Files))
	for _, f := range m.Files {
		shas = append(shas, f.SHA256)
	}
	have, err := s.Store.HasBlobs(ctx, shas)
	if err != nil {
		return nil, apierr.Internal(err)
	}
	missing := []string{}
	seen := map[string]bool{}
	for _, sha := range shas {
		if !have[sha] && !seen[sha] {
			missing = append(missing, sha)
			seen[sha] = true
		}
	}
	return missing, nil
}

// readPublish parses the multipart body: manifest first, then blob parts keyed by sha.
// Blob parts are written to blob storage as they stream, so the returned set is what was uploaded.
func (s *server) readPublish(w http.ResponseWriter, r *http.Request) (manifest, map[string]bool, *apierr.Error) {
	r.Body = http.MaxBytesReader(w, r.Body, s.Cfg.Limits.VersionBytes+1<<20)
	mr, err := r.MultipartReader()
	if err != nil {
		return manifest{}, nil, apierr.Invalid("body must be multipart/form-data")
	}
	first, err := mr.NextPart()
	if err != nil || first.FormName() != "manifest" {
		return manifest{}, nil, apierr.Invalid("first part must be named manifest")
	}
	raw, err := io.ReadAll(io.LimitReader(first, 1<<20))
	if err != nil {
		return manifest{}, nil, apierr.Invalid("manifest unreadable")
	}
	m, e := parseManifest(raw, s.Cfg.Limits)
	if e != nil {
		return m, nil, e
	}
	sizes := map[string]int64{}
	for _, f := range m.Files {
		sizes[f.SHA256] = f.Size
	}
	uploaded := map[string]bool{}
	for {
		p, err := mr.NextPart()
		if err == io.EOF {
			break
		}
		if err != nil {
			return m, nil, apierr.Invalid("malformed multipart body")
		}
		if e := s.storePart(r.Context(), p, sizes, uploaded); e != nil {
			return m, nil, e
		}
	}
	return m, uploaded, nil
}

func (s *server) storePart(ctx context.Context, p *multipart.Part, sizes map[string]int64, uploaded map[string]bool) *apierr.Error {
	sha := p.FormName()
	size, ok := sizes[sha]
	if !ok {
		return apierr.Invalid("part " + sha + " is not in the manifest")
	}
	if uploaded[sha] {
		io.Copy(io.Discard, p)
		return nil
	}
	if err := s.Blobs.Put(ctx, sha, p, size); err != nil {
		if errors.Is(err, blob.ErrChecksum) {
			return apierr.Invalid("checksum mismatch for part " + sha)
		}
		return apierr.Internal(err)
	}
	if err := s.Store.UpsertBlob(ctx, sha, size); err != nil {
		return apierr.Internal(err)
	}
	uploaded[sha] = true
	return nil
}

// publishVersion verifies every manifest blob exists, then records the version.
func (s *server) publishVersion(ctx context.Context, a store.Artifact, m manifest) (int, *apierr.Error) {
	missing, e := s.missingBlobs(ctx, m)
	if e != nil {
		return 0, e
	}
	if len(missing) > 0 {
		return 0, apierr.Invalid(fmt.Sprintf("missing blob %s (and %d more): upload it in this request", missing[0], len(missing)-1))
	}
	files := make([]store.File, 0, len(m.Files))
	var total int64
	for _, f := range m.Files {
		files = append(files, store.File{Path: f.Path, SHA256: f.SHA256, Size: f.Size, ContentType: contentTypeFor(f.Path)})
		total += f.Size
	}
	v := store.Version{ArtifactID: a.ID, Version: a.CurrentVersion + 1, CreatedAt: s.Now(), Bytes: total, FileCount: len(files)}
	if err := s.Store.AddVersion(ctx, v, files); err != nil {
		return 0, apierr.Internal(err)
	}
	if m.Title != "" && m.Title != a.Title {
		if err := s.Store.UpdateTitle(ctx, a.ID, m.Title); err != nil {
			return 0, apierr.Internal(err)
		}
	}
	if _, err := s.Store.PruneVersions(ctx, a.ID, s.Cfg.Limits.VersionsKept); err != nil {
		return 0, apierr.Internal(err)
	}
	return v.Version, nil
}
```

```go
package api

import (
	"context"
	"log/slog"
	"time"
)

// RunGC deletes zero-ref blobs older than 24h on a timer until ctx is done.
func RunGC(ctx context.Context, d Deps, every time.Duration) {
	t := time.NewTicker(every)
	defer t.Stop()
	for {
		select {
		case <-ctx.Done():
			return
		case <-t.C:
			if err := gcOnce(ctx, d, d.Now().Add(-24*time.Hour)); err != nil {
				slog.Error("gc", "err", err)
			}
		}
	}
}

func gcOnce(ctx context.Context, d Deps, olderThan time.Time) error {
	shas, err := d.Store.ZeroRefBlobs(ctx, olderThan)
	if err != nil {
		return err
	}
	for _, sha := range shas {
		if err := d.Blobs.Delete(ctx, sha); err != nil {
			return err
		}
		if err := d.Store.DeleteBlob(ctx, sha); err != nil {
			return err
		}
	}
	return nil
}
```

```go
// Package serve streams artifact versions to browsers with the comments runtime injected.
package serve

import (
	"bytes"
	"fmt"
)

func tag(artifactID string, version int) []byte {
	return []byte(fmt.Sprintf(`<script src="/_rt/comments.js" data-artifact="%s" data-version="%d" defer></script>`, artifactID, version))
}

// Inject places the runtime tag before the last </body> (any case) or appends it.
func Inject(html []byte, artifactID string, version int) []byte {
	lower := bytes.ToLower(html)
	i := bytes.LastIndex(lower, []byte("</body>"))
	t := tag(artifactID, version)
	if i < 0 {
		return append(append([]byte{}, html...), t...)
	}
	out := make([]byte, 0, len(html)+len(t))
	out = append(out, html[:i]...)
	out = append(out, t...)
	return append(out, html[i:]...)
}
```

```go
package serve

import (
	"crypto/sha256"
	"encoding/hex"
	"errors"
	"io"
	"net/http"
	"strconv"
	"strings"

	"github.com/anup-a/byagent/internal/apierr"
	"github.com/anup-a/byagent/internal/blob"
	"github.com/anup-a/byagent/internal/store"
)

type Handler struct {
	Store       store.Store
	Blobs       blob.Store
	Runtime     []byte
	ShareOrigin string
	runtimeETag string
}

func (h *Handler) Register(mux *http.ServeMux) {
	sum := sha256.Sum256(h.Runtime)
	h.runtimeETag = `"` + hex.EncodeToString(sum[:]) + `"`
	mux.HandleFunc("GET /_rt/comments.js", h.runtime)
	mux.HandleFunc("GET /a/{id}/v/{version}/{path...}", h.pinned)
	mux.HandleFunc("GET /a/{id}/{path...}", h.current)
	mux.HandleFunc("GET /a/{id}", func(w http.ResponseWriter, r *http.Request) {
		http.Redirect(w, r, "/a/"+r.PathValue("id")+"/", http.StatusMovedPermanently)
	})
}

func (h *Handler) runtime(w http.ResponseWriter, r *http.Request) {
	if r.Header.Get("If-None-Match") == h.runtimeETag {
		w.WriteHeader(304)
		return
	}
	w.Header().Set("Content-Type", "text/javascript; charset=utf-8")
	w.Header().Set("Cache-Control", "public, max-age=31536000, immutable")
	w.Header().Set("ETag", h.runtimeETag)
	w.Write(h.Runtime)
}

func (h *Handler) current(w http.ResponseWriter, r *http.Request) {
	h.serve(w, r, r.PathValue("id"), 0, r.PathValue("path"))
}

func (h *Handler) pinned(w http.ResponseWriter, r *http.Request) {
	v, err := strconv.Atoi(r.PathValue("version"))
	if err != nil || v < 1 {
		apierr.Write(w, r, apierr.NotFound("version"))
		return
	}
	h.serve(w, r, r.PathValue("id"), v, r.PathValue("path"))
}

func (h *Handler) csp() string {
	return "default-src 'self' data: blob:; script-src 'self' 'unsafe-inline' https://cdnjs.cloudflare.com https://cdn.jsdelivr.net; " +
		"style-src 'self' 'unsafe-inline' https://fonts.googleapis.com; font-src 'self' data: https://fonts.gstatic.com; " +
		"img-src 'self' data: blob: https:; connect-src 'self' " + h.ShareOrigin + "; frame-ancestors 'none'"
}

// serve resolves the file for (id, version, path) and streams it. version 0 = current.
func (h *Handler) serve(w http.ResponseWriter, r *http.Request, id string, version int, p string) {
	a, err := h.Store.Artifact(r.Context(), id)
	if errors.Is(err, store.ErrNotFound) {
		apierr.Write(w, r, apierr.NotFound("artifact"))
		return
	}
	if err != nil {
		apierr.Write(w, r, apierr.Internal(err))
		return
	}
	if a.DeletedAt != nil {
		apierr.Write(w, r, apierr.Gone())
		return
	}
	pinnedReq := version > 0
	if version == 0 {
		version = a.CurrentVersion
	}
	if p == "" || strings.HasSuffix(p, "/") {
		p += "index.html"
	}
	files, err := h.Store.Files(r.Context(), id, version)
	if err != nil {
		apierr.Write(w, r, apierr.Internal(err))
		return
	}
	var f *store.File
	for i := range files {
		if files[i].Path == p {
			f = &files[i]
			break
		}
	}
	if f == nil {
		apierr.Write(w, r, apierr.NotFound("file"))
		return
	}
	isHTML := strings.HasPrefix(f.ContentType, "text/html")
	etag := `"` + f.SHA256 + `"`
	if isHTML {
		etag = `"` + f.SHA256 + "-rt-" + strconv.Itoa(version) + `"`
	}
	w.Header().Set("Content-Security-Policy", h.csp())
	w.Header().Set("X-Content-Type-Options", "nosniff")
	w.Header().Set("ETag", etag)
	if pinnedReq {
		w.Header().Set("Cache-Control", "public, max-age=31536000, immutable")
	} else {
		w.Header().Set("Cache-Control", "public, max-age=60")
	}
	if r.Header.Get("If-None-Match") == etag {
		w.WriteHeader(304)
		return
	}
	rc, size, err := h.Blobs.Get(r.Context(), f.SHA256)
	if err != nil {
		apierr.Write(w, r, apierr.Internal(err))
		return
	}
	defer rc.Close()
	w.Header().Set("Content-Type", f.ContentType)
	if !isHTML {
		w.Header().Set("Content-Length", strconv.FormatInt(size, 10))
		io.Copy(w, rc)
		return
	}
	body, err := io.ReadAll(rc)
	if err != nil {
		apierr.Write(w, r, apierr.Internal(err))
		return
	}
	w.Write(Inject(body, id, version))
}
```

```go
	CreatedAt  time.Time `json:"created_at"`
}

type threadView struct {
	ID           string          `json:"id"`
	Version      int             `json:"version"`
	Status       string          `json:"status"`
	AnchorStatus string          `json:"anchor_status"`
	Anchor       json.RawMessage `json:"anchor"`
	CreatedAt    time.Time       `json:"created_at"`
	ResolvedAt   *time.Time      `json:"resolved_at"`
	Comments     []commentView   `json:"comments"`
}

func toThreadViews(ts []store.Thread, cs map[string][]store.Comment) []threadView {
	out := make([]threadView, 0, len(ts))
	for _, t := range ts {
		v := threadView{ID: t.ID, Version: t.Version, Status: t.Status, AnchorStatus: t.AnchorStatus, Anchor: json.RawMessage(t.AnchorJSON), CreatedAt: t.CreatedAt, ResolvedAt: t.ResolvedAt, Comments: []commentView{}}
		for _, c := range cs[t.ID] {
			v.Comments = append(v.Comments, commentView{c.ID, c.AuthorName, c.AuthorKind, c.Body, c.CreatedAt})
		}
		out = append(out, v)
	}
	return out
}

func (s *server) threadViews(r *http.Request, ts []store.Thread) ([]threadView, error) {
	idsList := make([]string, 0, len(ts))
	for _, t := range ts {
		idsList = append(idsList, t.ID)
	}
	cs, err := s.Store.Comments(r.Context(), idsList)
	if err != nil {
		return nil, err
	}
	return toThreadViews(ts, cs), nil
}

func (s *server) threadView(r *http.Request, id string) (threadView, *apierr.Error) {
	t, err := s.Store.Thread(r.Context(), id)
	if errors.Is(err, store.ErrNotFound) {
		return threadView{}, apierr.NotFound("thread")
	}
	if err != nil {
		return threadView{}, apierr.Internal(err)
	}
	vs, err := s.threadViews(r, []store.Thread{t})
	if err != nil {
		return threadView{}, apierr.Internal(err)
	}
	return vs[0], nil
}

func (s *server) validName(n string) (string, *apierr.Error) {
	n = strings.TrimSpace(n)
	if n == "" || utf8.RuneCountInString(n) > 80 {
		return "", apierr.Invalid("author_name must be 1..80 characters")
	}
	return n, nil
}

func (s *server) validBody(b string) (string, *apierr.Error) {
	b = strings.TrimSpace(b)
	if b == "" || len(b) > s.Cfg.Limits.CommentBytes {
		return "", apierr.Invalid("body must be 1.." + itoa(s.Cfg.Limits.CommentBytes) + " bytes")
	}
	return b, nil
}

func (s *server) validAnchor(raw json.RawMessage) (string, *apierr.Error) {
	if len(raw) == 0 || len(raw) > s.Cfg.Limits.AnchorBytes {
		return "", apierr.Invalid("anchor must be a JSON object up to " + itoa(s.Cfg.Limits.AnchorBytes) + " bytes")
	}
	var a struct {
		Kind  string `json:"kind"`
		Quote string `json:"quote"`
		CSS   string `json:"css"`
	}
	if err := json.Unmarshal(raw, &a); err != nil {
		return "", apierr.Invalid("anchor must be a JSON object")
	}
	switch a.Kind {
	case "text":
		if a.Quote == "" {
			return "", apierr.Invalid("text anchor requires quote")
		}
	case "element":
		if a.CSS == "" {
			return "", apierr.Invalid("element anchor requires css")
		}
	default:
		return "", apierr.Invalid("anchor.kind must be text or element")
	}
	return string(raw), nil
}

// loadArtifactForViewer returns the artifact for /c/* routes (no ownership check).
func (s *server) loadArtifactForViewer(r *http.Request, id string) (store.Artifact, *apierr.Error) {
	a, err := s.Store.Artifact(r.Context(), id)
	if errors.Is(err, store.ErrNotFound) {
		return a, apierr.NotFound("artifact")
	}
	if err != nil {
		return a, apierr.Internal(err)
	}
	if a.DeletedAt != nil {
		return a, apierr.Gone()
	}
	return a, nil
}

func (s *server) versionExists(r *http.Request, artifactID string, v int) (bool, error) {
	vs, err := s.Store.Versions(r.Context(), artifactID)
	if err != nil {
		return false, err
	}
	for _, x := range vs {
		if x.Version == v {
			return true, nil
		}
	}
	return false, nil
}

func decodeJSON(w http.ResponseWriter, r *http.Request, limit int64, dst any) *apierr.Error {
	if err := json.NewDecoder(http.MaxBytesReader(w, r.Body, limit)).Decode(dst); err != nil {
		return apierr.Invalid("body must be valid JSON")
	}
	return nil
}

// idempotency remembers viewer thread creations for 10 minutes, per process.
type idempotency struct {
	mu    sync.Mutex
	items map[string]idemEntry
}

type idemEntry struct {
	threadID string
	at       time.Time
}

func (i *idempotency) get(key string, now time.Time) (string, bool) {
	i.mu.Lock()
	defer i.mu.Unlock()
	for k, e := range i.items {
		if now.Sub(e.at) > 10*time.Minute {
			delete(i.items, k)
		}
	}
	e, ok := i.items[key]
	return e.threadID, ok
}

func (i *idempotency) put(key, threadID string, now time.Time) {
	i.mu.Lock()
	defer i.mu.Unlock()
	if i.items == nil {
		i.items = map[string]idemEntry{}
	}
	i.items[key] = idemEntry{threadID, now}
}
```

```go
func (s *server) viewerCreateThread(w http.ResponseWriter, r *http.Request) {
	if !s.viewerLimit(w, r) {
		return
	}
	a, e := s.loadArtifactForViewer(r, r.PathValue("id"))
	if e != nil {
		apierr.Write(w, r, e)
		return
	}
	var in struct {
		AuthorName string          `json:"author_name"`
		Body       string          `json:"body"`
		Anchor     json.RawMessage `json:"anchor"`
		Version    int             `json:"version"`
	}
	if e := decodeJSON(w, r, int64(s.Cfg.Limits.AnchorBytes+s.Cfg.Limits.CommentBytes+1024), &in); e != nil {
		apierr.Write(w, r, e)
		return
	}
	name, e := s.validName(in.AuthorName)
	if e != nil {
		apierr.Write(w, r, e)
		return
	}
	body, e := s.validBody(in.Body)
	if e != nil {
		apierr.Write(w, r, e)
		return
	}
	anchor, e := s.validAnchor(in.Anchor)
	if e != nil {
		apierr.Write(w, r, e)
		return
	}
	if ok, err := s.versionExists(r, a.ID, in.Version); err != nil {
		apierr.Write(w, r, apierr.Internal(err))
		return
	} else if !ok {
		apierr.Write(w, r, apierr.Invalid("version does not exist"))
		return
	}
	idemKey := r.Header.Get("Idempotency-Key")
	if len(idemKey) > 128 {
		apierr.Write(w, r, apierr.Invalid("Idempotency-Key too long"))
		return
	}
	now := s.Now()
	if idemKey != "" {
		if tid, ok := s.idem.get(a.ID+"|"+idemKey, now); ok {
			v, e := s.threadView(r, tid)
			if e != nil {
				apierr.Write(w, r, e)
				return
			}
			apierr.WriteJSON(w, 201, v)
			return
		}
	}
	th := store.Thread{ID: newThreadID(), ArtifactID: a.ID, Version: in.Version, AnchorJSON: anchor, Status: "open", AnchorStatus: "unknown", CreatedAt: now}
	c := store.Comment{ID: newCommentID(), ThreadID: th.ID, AuthorName: name, AuthorKind: "viewer", Body: body, CreatedAt: now}
	if err := s.Store.CreateThread(r.Context(), th, c); err != nil {
		apierr.Write(w, r, apierr.Internal(err))
		return
	}
	if idemKey != "" {
		s.idem.put(a.ID+"|"+idemKey, th.ID, now)
	}
	v, e := s.threadView(r, th.ID)
	if e != nil {
		apierr.Write(w, r, e)
		return
	}
	apierr.WriteJSON(w, 201, v)
}

		return

	if e != nil {
		apierr.Write(w, r, e)
		return
	}

		Version      int    `json:"version"`
		AnchorStatus string `json:"anchor_status"`

		return

		return

		status = ""
	}

	if v := r.URL.Query().Get("version"); v != "" {
		n, err := strconv.Atoi(v)

		return
	}

	if err != nil {
		apierr.Write(w, r, apierr.Internal(err))
		return
	}
	w.Header().Set("Cache-Control", "no-store")
	apierr.WriteJSON(w, 200, map[string]any{"threads": views})
}
```

```go
		apierr.Write(w, r, apierr.Internal(err))
		return
	}

		return
```
